cmake_minimum_required(VERSION 3.24)
project(OneBillion VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
option(ONEBILLION_BUILD_GAME "Build the SDL application" ON)
option(ONEBILLION_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(ONEBILLION_FETCH_SDL "Download the exact SDL release when not installed" ON)
include(CTest)

add_library(siege_core src/simulation.cpp)
target_include_directories(siege_core PUBLIC src)
if(MSVC)
  target_compile_options(siege_core PUBLIC /W4 /permissive- /utf-8)
else()
  target_compile_options(siege_core PUBLIC -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(ONEBILLION_SANITIZERS AND NOT MSVC)
  target_compile_options(siege_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(siege_core PUBLIC -fsanitize=address,undefined)
endif()

add_executable(siege_benchmark src/benchmark.cpp)
target_link_libraries(siege_benchmark PRIVATE siege_core)
if(BUILD_TESTING)
  add_executable(siege_tests tests/simulation_tests.cpp)
  target_link_libraries(siege_tests PRIVATE siege_core)
  add_test(NAME simulation COMMAND siege_tests)
endif()

if(ONEBILLION_BUILD_GAME)
  find_package(SDL3 3.4.18 EXACT QUIET CONFIG)
  if(NOT SDL3_FOUND)
    if(NOT ONEBILLION_FETCH_SDL)
      message(FATAL_ERROR "SDL 3.4.18 is required. Set SDL3_DIR or enable ONEBILLION_FETCH_SDL.")
    endif()
    include(FetchContent)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
    set(SDL_TESTS OFF CACHE BOOL "" FORCE)
    set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(SDL3
      URL https://www.libsdl.org/release/SDL3-3.4.18.tar.gz
      URL_HASH SHA256=9c75cf16330322c217dedd2e0609f1124f1b54b8633e763467b4684d0f4334a3
      DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(SDL3)
  endif()
  add_executable(one_billion src/main.cpp src/presentation.cpp)
  target_link_libraries(one_billion PRIVATE siege_core SDL3::SDL3)
  if(BUILD_TESTING)
    add_test(NAME ui_smoke COMMAND one_billion --self-test-ui)
    set_tests_properties(ui_smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
  endif()
  if(WIN32 AND TARGET SDL3::SDL3-shared)
    add_custom_command(TARGET one_billion POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different
        $<TARGET_FILE:SDL3::SDL3-shared> $<TARGET_FILE_DIR:one_billion>)
  endif()
  install(TARGETS one_billion siege_benchmark RUNTIME DESTINATION .)
endif()
